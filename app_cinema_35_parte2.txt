class TorrentStreamInfo {
  final String playUrl;
  final String fileName;
  final int fileSizeBytes;
  final int activePeers;
  final int totalSeeds;
  final double downloadSpeedBytes;
  final bool viaCloudCache;

  const TorrentStreamInfo({
    required this.playUrl,
    required this.fileName,
    required this.fileSizeBytes,
    required this.activePeers,
    required this.totalSeeds,
    required this.downloadSpeedBytes,
    this.viaCloudCache = false,
  });
}

class TorrServerService {
  static const MethodChannel _channel = MethodChannel('com.chiavegatti.cinema/torrserver');
  static const String baseUrl = 'http://127.0.0.1:8090';
  static Process? _process;
  static bool _configurado = false;
  static Future<bool>? _iniciandoFuture;
  static String ultimoErro = '';

  static void encerrar() {
    try {
      _process?.kill();
      _process = null;
      _configurado = false;
    } catch (_) {}
  }

  static Future<bool> _portaAberta() async {
    try {
      final s = await Socket.connect('127.0.0.1', 8090, timeout: const Duration(milliseconds: 350));
      s.destroy();
      return true;
    } catch (_) {
      return false;
    }
  }

  static Future<void> _aplicarConfiguracoes() async {
    if (_configurado) return;
    try {
      await _postJson('/settings', {
        'action': 'set',
        'sets': {
          'CacheSize': 16777216,
          'ReaderReadAHead': 95,
          'PreloadCache': 2,
          'UseDisk': false,
          'ForceEncrypt': false,
          'RetrackersMode': 1,
          'TorrentDisconnectTimeout': 30,
          'EnableIPv6': true,
          'DisableTCP': false,
          'DisableUTP': false,
          'DisableUPNP': false,
          'DisableDHT': false,
          'DisablePEX': false,
          'DisableUpload': false,
          'ConnectionsLimit': 60,
          'PeersListenPort': 0,
        },
      });
      _configurado = true;
    } catch (_) {}
  }

  static Future<bool> garantirAtivo() {
    if (_iniciandoFuture != null) return _iniciandoFuture!;
    _iniciandoFuture = _garantirAtivoInterno().whenComplete(() => _iniciandoFuture = null);
    return _iniciandoFuture!;
  }

  static Future<bool> _garantirAtivoInterno() async {
    if (await _portaAberta()) {
      await _aplicarConfiguracoes();
      return true;
    }
    try {
      final paths = await _channel.invokeMapMethod('getPaths');
      final nativeLibDir = (paths?['nativeLibDir'] ?? '').toString();
      final filesDir = (paths?['filesDir'] ?? '').toString();
      final binPath = '$nativeLibDir/libtorrserver.so';
      if (nativeLibDir.isEmpty || !await File(binPath).exists()) {
        ultimoErro = 'Motor de streaming local não encontrado.';
        return false;
      }
      final dataDir = Directory('$filesDir/torrserver');
      if (!await dataDir.exists()) await dataDir.create(recursive: true);

      final tentativas = [
        ['-p', '8090', '-d', dataDir.path],
        ['--port', '8090', '--path', dataDir.path],
        ['']..clear(),
      ];

      for (final args in tentativas) {
        _process?.kill();
        _process = await Process.start(
          binPath,
          args,
          workingDirectory: dataDir.path,
          environment: {
            'HOME': filesDir,
            'TMPDIR': dataDir.path,
            'RES_OPTIONS': 'ndots:0',
            'GODEBUG': 'netdns=go',
          },
        );
        _process!.stdout.listen((_) {}, onError: (_) {});
        _process!.stderr.listen((_) {}, onError: (_) {});
        for (int i = 0; i < 16; i++) {
          await Future.delayed(const Duration(milliseconds: 200));
          if (await _portaAberta()) {
            ultimoErro = '';
            await _aplicarConfiguracoes();
            return true;
          }
        }
      }
      ultimoErro = 'Tempo esgotado ao iniciar o motor local.';
      return false;
    } catch (e) {
      ultimoErro = 'Erro ao iniciar motor local: $e';
      return false;
    }
  }

  static Future _postJson(String endpoint, Map body) async {
    final c = HttpClient()..connectionTimeout = const Duration(seconds: 4);
    try {
      final req = await c.postUrl(Uri.parse('$baseUrl$endpoint'));
      req.headers.contentType = ContentType.json;
      req.write(jsonEncode(body));
      final res = await req.close().timeout(const Duration(seconds: 5));
      final txt = await utf8.decoder.bind(res).join();
      if (txt.trim().startsWith('{')) return jsonDecode(txt);
    } catch (_) {} finally {
      c.close(force: true);
    }
    return null;
  }

  static Future<void> removerTorrent(String magnetOuHash) async {
    try {
      final hash = magnetOuHash.length == 40 ? magnetOuHash.toLowerCase() : _extrairHash(magnetOuHash);
      if (hash.length == 40) {
        await _postJson('/torrents', {'action': 'rem', 'hash': hash});
      }
    } catch (_) {}
  }

  static Future<List<int>?> _baixarTorrentCache(String hashHex) async {
    final hUpper = hashHex.toUpperCase();
    final urls = [
      'https://itorrents.org/torrent/$hUpper.torrent',
      'https://torrage.info/torrent.php?h=$hUpper',
      'https://btcache.me/torrent/$hUpper',
    ];
    final completer = Completer<List<int>?>();
    int pendentes = urls.length;
    for (final u in urls) {
      () async {
        final c = HttpClient()..connectionTimeout = const Duration(seconds: 4);
        try {
          final req = await c.getUrl(Uri.parse(u));
          req.headers.set('User-Agent', 'Mozilla/5.0');
          final res = await req.close().timeout(const Duration(seconds: 5));
          if (res.statusCode == 200) {
            final bytes = await res.fold<List<int>>([], (prev, el) => prev..addAll(el));
            if (bytes.length > 100 && bytes.first == 100 && !completer.isCompleted) {
              completer.complete(bytes);
              return;
            }
          } else {
            await res.drain();
          }
        } catch (_) {} finally {
          c.close(force: true);
          pendentes--;
          if (pendentes == 0 && !completer.isCompleted) completer.complete(null);
        }
      }();
    }
    return completer.future;
  }

  static Future<bool> _uploadTorrentParaServidor(List<int> bytes, String hashHex) async {
    final c = HttpClient()..connectionTimeout = const Duration(seconds: 4);
    try {
      const boundary = '----TorrServerBoundary7MA4YWxkTrZu0gW';
      final req = await c.postUrl(Uri.parse('$baseUrl/torrent/upload'));
      req.headers.set('Content-Type', 'multipart/form-data; boundary=$boundary');
      final header = '--$boundary\r\n'
          'Content-Disposition: form-data; name="file"; filename="$hashHex.torrent"\r\n'
          'Content-Type: application/x-bittorrent\r\n\r\n';
      const footer = '\r\n--$boundary--\r\n';
      req.add(utf8.encode(header));
      req.add(bytes);
      req.add(utf8.encode(footer));
      final res = await req.close().timeout(const Duration(seconds: 5));
      await res.drain();
      return res.statusCode >= 200 && res.statusCode < 300;
    } catch (_) {
      return false;
    } finally {
      c.close(force: true);
    }
  }

  static String _base32ParaHex(String b32) {
    const alpha = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';
    final clean = b32.toUpperCase().replaceAll('=', '');
    int bits = 0, val = 0;
    final bytes = [];
    for (int i = 0; i < clean.length; i++) {
      final idx = alpha.indexOf(clean[i]);
      if (idx < 0) return b32.toLowerCase();
      val = (val << 5) | idx;
      bits += 5;
      if (bits >= 8) {
        bytes.add((val >> (bits - 8)) & 0xFF);
        bits -= 8;
      }
    }
    return bytes.map((b) => (b as int).toRadixString(16).padLeft(2, '0')).join();
  }

  static String _extrairHash(String magnet) {
    final m = RegExp(r'btih:([a-zA-Z0-9]{32,40})', caseSensitive: false).firstMatch(magnet);
    if (m == null) return '';
    final raw = m.group(1)!;
    return raw.length == 32 ? _base32ParaHex(raw) : raw.toLowerCase();
  }

  static String _enriquecerMagnet(String magnet) {
    var clean = magnet.trim().replaceFirstMapped(
      RegExp(r'(btih:)([a-zA-Z2-7]{32})(?=&|$)', caseSensitive: false),
      (m) => '${m.group(1)}${_base32ParaHex(m.group(2)!)}',
    );
    const trackers = [
      'udp://tracker.opentrackr.org:1337/announce',
      'http://tracker.opentrackr.org:1337/announce',
      'udp://open.stealth.si:80/announce',
      'udp://tracker.torrent.eu.org:451/announce',
      'udp://open.demonii.com:1337/announce',
      'udp://tracker.openbittorrent.com:6969/announce',
      'http://tracker.openbittorrent.com:80/announce',
      'udp://exodus.desync.com:6969/announce',
      'udp://tracker.moeking.me:6969/announce',
      'udp://explodie.org:6969/announce',
      'udp://tracker.tiny-vps.com:6969/announce',
      'udp://tracker.dler.org:6969/announce',
      'https://tracker.tamersunion.org:443/announce',
      'https://tracker.loligirl.cn:443/announce',
      'https://tracker.gbitt.info:443/announce',
      'http://93.158.213.92:1337/announce',
      'udp://93.158.213.92:1337/announce',
      'udp://tracker1.bt.moack.co.kr:80/announce',
      'udp://tracker.theoks.net:6969/announce',
      'udp://tracker.bittor.pw:1337/announce',
      'udp://movies.zsw.ca:6969/announce',
      'udp://tracker.cyberia.is:6969/announce',
      'udp://retracker01-msk-virt.corbina.net:80/announce',
      'udp://p4p.arenabg.com:1337/announce',
      'udp://tracker.leechers-paradise.org:6969/announce',
      'udp://tracker.coppersurfer.tk:6969/announce',
      'udp://9.rarbg.com:2810/announce',
      'udp://tracker.publictracker.xyz:6969/announce',
    ];
    final sb = StringBuffer(clean);
    final lower = clean.toLowerCase();
    for (final tr in trackers) {
      final enc = Uri.encodeComponent(tr).toLowerCase();
      if (!lower.contains(enc) && !lower.contains(tr.toLowerCase())) {
        sb.write('&tr=${Uri.encodeComponent(tr)}');
      }
    }
    return sb.toString();
  }

  static Future<TorrentStreamInfo?> prepararMagnet(
    String magnetUrl, {
    void Function(String)? onStatus,
    void Function(double)? onProgress,
    bool Function()? cancelado,
  }) async {
    ultimoErro = '';
    if (!magnetUrl.trim().toLowerCase().startsWith('magnet:?xt=urn:btih:')) {
      ultimoErro = 'Magnet negado: ausente prefixo obrigatório magnet:?xt=urn:btih:';
      return null;
    }
    final cleanMagnet = _enriquecerMagnet(magnetUrl);
    final hash = _extrairHash(cleanMagnet);
    onProgress?.call(0.12);
    onStatus?.call('Iniciando motor de vídeo...');
    if (!await garantirAtivo()) return null;
    if (cancelado != null && cancelado()) return null;

    onProgress?.call(0.25);
    onStatus?.call('Verificando fontes e trackers...');
    bool subiuTorrent = false;
    if (hash.length == 40) {
      final torrentBytes = await _baixarTorrentCache(hash);
      if (torrentBytes != null) {
        subiuTorrent = await _uploadTorrentParaServidor(torrentBytes, hash);
      }
    }

    await _postJson('/torrents', {
      'action': 'add',
      'link': cleanMagnet,
      'title': 'Stream',
      'save_to_db': false,
    });

    int fileId = 1;
    String fileName = 'video.mkv';
    int fileSize = 0;
    int activePeers = 0;
    int totalSeeds = 0;
    double speedBytes = 0;
    bool encontrouArquivo = false;

    for (int i = 0; i < 30; i++) {
      if (cancelado != null && cancelado()) return null;
      await Future.delayed(const Duration(milliseconds: 500));
      onProgress?.call((0.30 + (i * 0.018)).clamp(0.30, 0.88));
      final info = await _postJson('/torrents', {'action': 'get', 'hash': hash});
      if (info is Map) {
        activePeers = (info['active_peers'] is num) ? (info['active_peers'] as num).toInt() : 0;
        totalSeeds = (info['connected_seeders'] is num)
            ? (info['connected_seeders'] as num).toInt()
            : ((info['total_peers'] is num) ? (info['total_peers'] as num).toInt() : 0);
        speedBytes = (info['download_speed'] is num) ? (info['download_speed'] as num).toDouble() : 0.0;

        final files = info['file_stats'];
        if (files is List && files.isNotEmpty) {
          Map? maior;
          for (final f in files) {
            if (f is Map) {
              final p = (f['path'] ?? '').toString().toLowerCase();
              final ehVid = p.endsWith('.mkv') || p.endsWith('.mp4') || p.endsWith('.avi') || p.endsWith('.ts');
              if (ehVid && (maior == null || (f['length'] ?? 0) > (maior['length'] ?? 0))) {
                maior = f;
              }
            }
          }
          maior ??= files.first as Map;
          fileId = (maior['id'] is int) ? maior['id'] as int : 1;
          fileName = (maior['path'] ?? 'video.mkv').toString().split('/').last;
          fileSize = (maior['length'] is int) ? maior['length'] as int : 0;
          encontrouArquivo = true;
          if (activePeers > 0 || totalSeeds > 0 || i >= 8) break;
        }
      }
      if (i >= 14 && !encontrouArquivo && activePeers == 0 && totalSeeds == 0) break;
      onStatus?.call('Buscando peers ativos (peers: $activePeers)...');
    }

    if (!encontrouArquivo || (activePeers == 0 && totalSeeds == 0 && !subiuTorrent)) {
      ultimoErro = 'Fonte sem peers ativos no momento.';
      return null;
    }

    onProgress?.call(0.94);
    onStatus?.call('Iniciando reprodução...');
    return TorrentStreamInfo(
      playUrl: '$baseUrl/play/$hash/$fileId?play',
      fileName: fileName,
      fileSizeBytes: fileSize,
      activePeers: activePeers,
      totalSeeds: totalSeeds,
      downloadSpeedBytes: speedBytes,
    );
  }
}

class TopIconeComTooltip extends StatefulWidget {
  final IconData icon;
  final String label;
  final VoidCallback onPressed;

  const TopIconeComTooltip({
    super.key,
    required this.icon,
    required this.label,
    required this.onPressed,
  });

  @override
  State<TopIconeComTooltip> createState() => _TopIconeComTooltipState();
}

class _TopIconeComTooltipState extends State<TopIconeComTooltip> {
  final LayerLink _layerLink = LayerLink();
  OverlayEntry? _overlayEntry;
  bool _focado = false;
  Timer? _timer;

  void _removerOverlay() {
    _overlayEntry?.remove();
    _overlayEntry = null;
  }

  void _mostrarOverlay() {
    _removerOverlay();
    _timer?.cancel();
    if (!mounted) return;
    final overlay = Overlay.maybeOf(context);
    if (overlay == null) return;
    _overlayEntry = OverlayEntry(
      builder: (_) => Positioned(
        width: 190,
        child: CompositedTransformFollower(
          link: _layerLink,
          showWhenUnlinked: false,
          targetAnchor: Alignment.bottomCenter,
          followerAnchor: Alignment.topCenter,
          offset: const Offset(0, 6),
          child: IgnorePointer(
            child: Material(
              color: Colors.transparent,
              child: Center(
                child: Container(
                  padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 5),
                  decoration: BoxDecoration(
                    color: const Color(0xFF141821).withOpacity(0.98),
                    borderRadius: BorderRadius.circular(6),
                    border: Border.all(color: const Color(0xFFF5B82E).withOpacity(0.85)),
                    boxShadow: const [
                      BoxShadow(color: Colors.black87, blurRadius: 10, offset: Offset(0, 4)),
                    ],
                  ),
                  child: Text(
                    widget.label,
                    textAlign: TextAlign.center,
                    style: const TextStyle(
                      color: Colors.white,
                      fontSize: 11.5,
                      fontWeight: FontWeight.w700,
                    ),
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    );
    overlay.insert(_overlayEntry!);
    _timer = Timer(const Duration(seconds: 2), () {
      if (mounted) _removerOverlay();
    });
  }

  @override
  void dispose() {
    _timer?.cancel();
    _removerOverlay();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return CompositedTransformTarget(
      link: _layerLink,
      child: CinemaFocusBox(
        onTap: widget.onPressed,
        borderRadius: 24,
        scaleOnFocus: 1.12,
        onFocusChanged: (hasFocus) {
          if (mounted) setState(() => _focado = hasFocus);
          if (hasFocus) {
            _mostrarOverlay();
          } else {
            _removerOverlay();
          }
        },
        child: AnimatedContainer(
          duration: const Duration(milliseconds: 160),
          padding: const EdgeInsets.all(9),
          decoration: BoxDecoration(
            color: _focado
                ? const Color(0xFFF5B82E).withOpacity(0.24)
                : Colors.white.withOpacity(0.08),
            shape: BoxShape.circle,
          ),
          child: Icon(
            widget.icon,
            size: 20,
            color: _focado ? const Color(0xFFF5B82E) : Colors.white,
          ),
        ),
      ),
    );
  }
}

class CinemaFocusBox extends StatefulWidget {
  final Widget child;
  final VoidCallback? onTap;
  final double borderRadius;
  final double scaleOnFocus;
  final bool autofocus;
  final EdgeInsetsGeometry padding;
  final ValueChanged<bool>? onFocusChanged;
  final FocusNode? focusNode;

  const CinemaFocusBox({
    super.key,
    required this.child,
    this.onTap,
    this.borderRadius = 12,
    this.scaleOnFocus = 1.045,
    this.autofocus = false,
    this.padding = EdgeInsets.zero,
    this.onFocusChanged,
    this.focusNode,
  });

  @override
  State<CinemaFocusBox> createState() => _CinemaFocusBoxState();
}

class _CinemaFocusBoxState extends State<CinemaFocusBox> {
  bool _focused = false;
  bool _hovered = false;
  DateTime _ultimoAcionamento = DateTime.fromMillisecondsSinceEpoch(0);

  bool get _ativo => _focused || _hovered;

  void _acionar() {
    if (widget.onTap == null) return;
    final agora = DateTime.now();
    if (agora.difference(_ultimoAcionamento).inMilliseconds < 280) return;
    _ultimoAcionamento = agora;
    widget.onTap!();
  }

  bool _isConfirmKey(LogicalKeyboardKey key) {
    return key == LogicalKeyboardKey.select ||
        key == LogicalKeyboardKey.enter ||
        key == LogicalKeyboardKey.numpadEnter ||
        key == LogicalKeyboardKey.accept ||
        key == LogicalKeyboardKey.gameButtonA ||
        key == LogicalKeyboardKey.space;
  }

  @override
  Widget build(BuildContext context) {
    return MouseRegion(
      cursor: widget.onTap != null ? SystemMouseCursors.click : MouseCursor.defer,
      onEnter: (_) { if (mounted) setState(() => _hovered = true); },
      onExit: (_) { if (mounted) setState(() => _hovered = false); },
      child: Actions(
        actions: <Type, Action<Intent>>{
          ActivateIntent: CallbackAction<ActivateIntent>(
            onInvoke: (_) {
              _acionar();
              return null;
            },
          ),
          ButtonActivateIntent: CallbackAction<ButtonActivateIntent>(
            onInvoke: (_) {
              _acionar();
              return null;
            },
          ),
        },
        child: Focus(
          focusNode: widget.focusNode,
          autofocus: widget.autofocus,
          onFocusChange: (f) {
            if (mounted) setState(() => _focused = f);
            widget.onFocusChanged?.call(f);
          },
          onKeyEvent: (node, event) {
            if (_isConfirmKey(event.logicalKey)) {
              if (event is KeyDownEvent) {
                _acionar();
              }
              return KeyEventResult.handled;
            }
            return KeyEventResult.ignored;
          },
          child: GestureDetector(
            onTap: _acionar,
            child: AnimatedScale(
              scale: _ativo ? widget.scaleOnFocus : 1.0,
              duration: const Duration(milliseconds: 140),
              curve: Curves.easeOutCubic,
              child: Container(
                padding: widget.padding,
                decoration: BoxDecoration(
                  borderRadius: BorderRadius.circular(widget.borderRadius),
                  border: Border.all(
                    color: _ativo ? const Color(0xFFF5B82E) : Colors.transparent,
                    width: 2.5,
                  ),
                ),
                child: widget.child,
              ),
            ),
          ),
        ),
      ),
    );
  }
}

AvatarTematico buscarAvatarPorId(String id) {
  for (final a in kAvataresCineMax) {
    if (a.id == id) return a;
  }
  return kAvataresCineMax.first;
}

class AvatarIlustradoWidget extends StatelessWidget {
  final String avatarId;
  final double size;
  final bool infantil;

  const AvatarIlustradoWidget({super.key, required this.avatarId, this.size = 76, this.infantil = false});

  @override
  Widget build(BuildContext context) {
    final av = buscarAvatarPorId(avatarId);
    return Stack(
      clipBehavior: Clip.none,
      children: [
        Container(
          width: size,
          height: size,
          alignment: Alignment.center,
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(size * 0.24),
            gradient: LinearGradient(begin: Alignment.topLeft, end: Alignment.bottomRight, colors: av.cores),
            boxShadow: [BoxShadow(color: av.cores.first.withOpacity(0.35), blurRadius: 10, offset: const Offset(0, 4))],
          ),
          clipBehavior: Clip.antiAlias,
          child: (av.imageUrl != null && av.imageUrl!.isNotEmpty)
              ? CachedNetworkImage(
                  imageUrl: av.imageUrl!,
                  width: size,
                  height: size,
                  fit: BoxFit.cover,
                  memCacheWidth: 160,
                  fadeInDuration: Duration.zero,
                  placeholder: (_, __) => Center(child: Text(av.emoji, style: TextStyle(fontSize: size * 0.48))),
                  errorWidget: (_, __, ___) => Center(child: Text(av.emoji, style: TextStyle(fontSize: size * 0.48))),
                )
              : Text(av.emoji, style: TextStyle(fontSize: size * 0.48)),
        ),
        if (infantil)
          Positioned(
            right: -4,
            bottom: -4,
            child: Container(
              padding: const EdgeInsets.symmetric(horizontal: 5, vertical: 2),
              decoration: BoxDecoration(
                color: const Color(0xFF00D2FF),
                borderRadius: BorderRadius.circular(6),
                border: Border.all(color: const Color(0xFF0A0C10), width: 1.5),
              ),
              child: const Text('KIDS', style: TextStyle(color: Colors.black, fontSize: 8.5, fontWeight: FontWeight.w900)),
            ),
          ),
      ],
    );
  }
}

class PerfisSelectionOverlay extends StatefulWidget {
  final VoidCallback? onPerfilSelecionado;
  final bool obrigatorio;

  const PerfisSelectionOverlay({super.key, this.onPerfilSelecionado, this.obrigatorio = false});

  static Future<void> abrir(BuildContext context, {VoidCallback? onPerfilSelecionado}) async {
    await MaxMemoryService.instance.inicializar();
    if (!context.mounted) return;
    await showDialog(
      context: context,
      barrierDismissible: true,
      barrierColor: Colors.black.withOpacity(0.88),
      builder: (_) => PerfisSelectionOverlay(onPerfilSelecionado: onPerfilSelecionado),
    );
  }

  @override
  State<PerfisSelectionOverlay> createState() => _PerfisSelectionOverlayState();
}

class _PerfisSelectionOverlayState extends State<PerfisSelectionOverlay> {
  bool _modoGerenciar = false;

  Future<void> _selecionar(MaxPerfilUsuario perfil) async {
    if (_modoGerenciar) {
      await _abrirFormPerfil(editar: perfil);
      return;
    }
    await MaxMemoryService.instance.definirPerfilAtivo(perfil.nome, infantil: perfil.infantil, avatarId: perfil.avatarId);
    widget.onPerfilSelecionado?.call();
    if (!widget.obrigatorio && mounted) Navigator.of(context).pop();
    if (perfil.nome.toLowerCase() != 'geral') {
      MaxAssistantService.instance.falar('Olá, ${perfil.nome}! O que vamos assistir hoje?');
    }
  }

  Future<void> _abrirFormPerfil({MaxPerfilUsuario? editar}) async {
    final ehEdicao = editar != null;
    final ehGeral = editar?.nome.toLowerCase() == 'geral';
    final nomeCtrl = TextEditingController(text: editar?.nome ?? '');
    bool ehInfantil = editar?.infantil ?? false;
    String categoriaFiltro = 'Todos';
    String avatarSelecionado = editar?.avatarId ?? 'geral_pipoca';
    const categorias = ['Todos', 'Filmes', 'Séries', 'Animes', 'Desenhos', 'Geral'];

    try {
    await showDialog(
      context: context,
      builder: (ctx) => StatefulBuilder(
        builder: (ctx, setLocal) {
          final avataresFiltrados = categoriaFiltro == 'Todos'
              ? kAvataresCineMax
              : kAvataresCineMax.where((a) => a.categoria == categoriaFiltro).toList();

          return AlertDialog(
            backgroundColor: const Color(0xFF141824),
            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(18), side: const BorderSide(color: Color(0xFFF5B82E), width: 1.4)),
            title: Text(
              ehEdicao ? 'Editar Perfil' : 'Novo Perfil CineMax',
              style: const TextStyle(color: Color(0xFFF5B82E), fontWeight: FontWeight.w900),
            ),
            content: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 420),
              child: SingleChildScrollView(
                child: Column(
                  mainAxisSize: MainAxisSize.min,
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    TextField(
                      controller: nomeCtrl,
                      enabled: !ehGeral,
                      autofocus: !ehEdicao,
                      style: const TextStyle(color: Colors.white),
                      decoration: InputDecoration(
                        hintText: 'Nome do perfil...',
                        hintStyle: const TextStyle(color: Colors.white38),
                        filled: true,
                        fillColor: const Color(0xFF1E2436),
                        border: OutlineInputBorder(borderRadius: BorderRadius.circular(12), borderSide: BorderSide.none),
                      ),
                    ),
                    const SizedBox(height: 8),
                    SwitchListTile(
                      contentPadding: EdgeInsets.zero,
                      activeColor: const Color(0xFFF5B82E),
                      title: const Text('Perfil Kids (filtra conteúdo infantil)', style: TextStyle(color: Colors.white, fontSize: 13.5)),
                      value: ehInfantil,
                      onChanged: (v) => setLocal(() {
                        ehInfantil = v;
                        if (v) avatarSelecionado = 'desenho_estrela';
                      }),
                    ),
                    const SizedBox(height: 8),
                    const Text('Escolha uma foto / avatar:', style: TextStyle(color: Colors.white70, fontSize: 13, fontWeight: FontWeight.w700)),
                    const SizedBox(height: 8),
                    SizedBox(
                      height: 34,
                      child: ListView.separated(
                        scrollDirection: Axis.horizontal,
                        itemCount: categorias.length,
                        separatorBuilder: (_, __) => const SizedBox(width: 6),
                        itemBuilder: (_, idx) {
                          final cat = categorias[idx];
                          final ativo = cat == categoriaFiltro;
                          return CinemaFocusBox(
                            borderRadius: 16,
                            onTap: () => setLocal(() => categoriaFiltro = cat),
                            child: Container(
                              padding: const EdgeInsets.symmetric(horizontal: 12, vertical: 6),
                              decoration: BoxDecoration(
                                color: ativo ? const Color(0xFFF5B82E) : Colors.white12,
                                borderRadius: BorderRadius.circular(16),
                              ),
                              alignment: Alignment.center,
                              child: Text(cat, style: TextStyle(fontSize: 11.5, fontWeight: FontWeight.w700, color: ativo ? Colors.black : Colors.white)),
                            ),
                          );
                        },
                      ),
                    ),
                    const SizedBox(height: 12),
                    Wrap(
                      spacing: 10,
                      runSpacing: 10,
                      children: avataresFiltrados.map((av) {
                        final sel = av.id == avatarSelecionado;
                        return CinemaFocusBox(
                          borderRadius: 16,
                          scaleOnFocus: 1.06,
                          onTap: () => setLocal(() => avatarSelecionado = av.id),
                          child: Container(
                            padding: const EdgeInsets.all(4),
                            decoration: BoxDecoration(
                              borderRadius: BorderRadius.circular(16),
                              border: Border.all(color: sel ? const Color(0xFFF5B82E) : Colors.transparent, width: 2.5),
                            ),
                            child: Column(
                              mainAxisSize: MainAxisSize.min,
                              children: [
                                AvatarIlustradoWidget(avatarId: av.id, size: 52),
                                const SizedBox(height: 3),
                                Text(av.nome, style: const TextStyle(color: Colors.white70, fontSize: 10)),
                              ],
                            ),
                          ),
                        );
                      }).toList(),
                    ),
                  ],
                ),
              ),
            ),
            actions: [
              if (ehEdicao && !ehGeral)
                CinemaFocusBox(
                  borderRadius: 10,
                  onTap: () async {
                    await MaxMemoryService.instance.excluirPerfil(editar!.nome);
                    if (ctx.mounted) Navigator.pop(ctx);
                    if (mounted) setState(() {});
                    widget.onPerfilSelecionado?.call();
                  },
                  child: const Padding(
                    padding: EdgeInsets.symmetric(horizontal: 10, vertical: 8),
                    child: Row(
                      mainAxisSize: MainAxisSize.min,
                      children: [
                        Icon(Icons.delete_outline, size: 18, color: Colors.redAccent),
                        SizedBox(width: 4),
                        Text('Excluir', style: TextStyle(color: Colors.redAccent, fontWeight: FontWeight.w800)),
                      ],
                    ),
                  ),
                ),
              CinemaFocusBox(
                borderRadius: 10,
                onTap: () => Navigator.pop(ctx),
                child: const Padding(
                  padding: EdgeInsets.symmetric(horizontal: 12, vertical: 8),
                  child: Text('Cancelar', style: TextStyle(color: Colors.white70, fontWeight: FontWeight.w700)),
                ),
              ),
              CinemaFocusBox(
                borderRadius: 10,
                onTap: () async {
                  final nome = nomeCtrl.text.trim();
                  if (nome.isEmpty) return;
                  if (ehEdicao) {
                    await MaxMemoryService.instance.editarPerfil(
                      nomeAntigo: editar!.nome,
                      novoNome: nome,
                      infantil: ehInfantil,
                      avatarId: avatarSelecionado,
                    );
                    if (ctx.mounted) Navigator.pop(ctx);
                    if (mounted) setState(() {});
                    widget.onPerfilSelecionado?.call();
                  } else {
                    await MaxMemoryService.instance.definirPerfilAtivo(nome, infantil: ehInfantil, avatarId: avatarSelecionado);
                    if (ctx.mounted) Navigator.pop(ctx);
                    if (mounted) setState(() {});
                    widget.onPerfilSelecionado?.call();
                    if (!widget.obrigatorio && mounted) Navigator.pop(context);
                  }
                },
                child: Container(
                  padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 9),
                  decoration: BoxDecoration(color: const Color(0xFFF5B82E), borderRadius: BorderRadius.circular(10)),
                  child: const Text('Salvar', style: TextStyle(color: Colors.black, fontWeight: FontWeight.w900)),
                ),
              ),
            ],
          );
        },
      ),
    );
    } finally {
      nomeCtrl.dispose();
    }
  }

  Widget _buildConteudo() {
    final perfis = MaxMemoryService.instance.perfis;
    return Center(
      child: SingleChildScrollView(
        padding: const EdgeInsets.all(28),
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            const CinemaLogoWidget(height: 46, width: 170),
            const SizedBox(height: 20),
            Text(
              _modoGerenciar ? 'Toque em um perfil para editar ou excluir' : 'Quem está assistindo?',
              textAlign: TextAlign.center,
              style: const TextStyle(color: Colors.white, fontSize: 22, fontWeight: FontWeight.w900),
            ),
            const SizedBox(height: 24),
            Wrap(
              spacing: 22,
              runSpacing: 22,
              alignment: WrapAlignment.center,
              children: [
                ...perfis.asMap().entries.map((entry) {
                  final p = entry.value;
                  return CinemaFocusBox(
                    autofocus: entry.key == 0,
                    borderRadius: 20,
                    scaleOnFocus: 1.08,
                    onTap: () => _selecionar(p),
                    child: Padding(
                      padding: const EdgeInsets.all(8),
                      child: Column(
                        mainAxisSize: MainAxisSize.min,
                        children: [
                          Stack(
                            clipBehavior: Clip.none,
                            children: [
                              AvatarIlustradoWidget(avatarId: p.avatarId, size: 88, infantil: p.infantil),
                              if (_modoGerenciar)
                                Positioned.fill(
                                  child: Container(
                                    decoration: BoxDecoration(
                                      color: Colors.black54,
                                      borderRadius: BorderRadius.circular(21),
                                    ),
                                    child: const Icon(Icons.edit_rounded, color: Color(0xFFF5B82E), size: 34),
                                  ),
                                ),
                            ],
                          ),
                          const SizedBox(height: 8),
                          Text(p.nome, style: const TextStyle(color: Colors.white, fontSize: 15, fontWeight: FontWeight.w800)),
                        ],
                      ),
                    ),
                  );
                }),
                CinemaFocusBox(
                  borderRadius: 20,
                  scaleOnFocus: 1.08,
                  onTap: () => _abrirFormPerfil(),
                  child: Padding(
                    padding: const EdgeInsets.all(8),
                    child: Column(
                      mainAxisSize: MainAxisSize.min,
                      children: [
                        Container(
                          width: 88,
                          height: 88,
                          decoration: BoxDecoration(
                            color: const Color(0xFF181D2B),
                            borderRadius: BorderRadius.circular(21),
                            border: Border.all(color: Colors.white30, width: 1.5),
                          ),
                          child: const Icon(Icons.add_rounded, color: Color(0xFFF5B82E), size: 42),
                        ),
                        const SizedBox(height: 8),
                        const Text('Adicionar Perfil', style: TextStyle(color: Colors.white70, fontSize: 14, fontWeight: FontWeight.w700)),
                      ],
                    ),
                  ),
                ),
              ],
            ),
            const SizedBox(height: 24),
            CinemaFocusBox(
              borderRadius: 22,
              scaleOnFocus: 1.05,
              onTap: () => setState(() => _modoGerenciar = !_modoGerenciar),
              child: Container(
                padding: const EdgeInsets.symmetric(horizontal: 20, vertical: 10),
                decoration: BoxDecoration(
                  color: _modoGerenciar ? const Color(0xFFF5B82E) : Colors.white10,
                  borderRadius: BorderRadius.circular(22),
                  border: Border.all(color: const Color(0xFFF5B82E)),
                ),
                child: Row(
                  mainAxisSize: MainAxisSize.min,
                  children: [
                    Icon(
                      _modoGerenciar ? Icons.check_rounded : Icons.manage_accounts_rounded,
                      color: _modoGerenciar ? Colors.black : const Color(0xFFF5B82E),
                      size: 19,
                    ),
                    const SizedBox(width: 8),
                    Text(
                      _modoGerenciar ? 'Concluir Edição' : 'Editar / Excluir Perfis',
                      style: TextStyle(
                        color: _modoGerenciar ? Colors.black : Colors.white,
                        fontWeight: FontWeight.w800,
                        fontSize: 13.5,
                      ),
                    ),
                  ],
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }

  @override
  Widget build(BuildContext context) {
    if (widget.obrigatorio) {
      return Scaffold(backgroundColor: const Color(0xFF0A0C10), body: SafeArea(child: _buildConteudo()));
    }
    return Dialog(
      backgroundColor: const Color(0xFF0E111A),
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(24), side: const BorderSide(color: Color(0xFFF5B82E), width: 1.5)),
      child: _buildConteudo(),
    );
  }
}
